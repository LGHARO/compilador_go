// Teste: if/while com string de entrada (ScanlnStr)
// Lê uma palavra da entrada e verifica se é "sim"
var resposta string = ScanlnStr()
if resposta == "sim" {
Println("Voce disse sim!")
} else {
Println("Voce nao disse sim.")
}

// Loop: lê strings até receber "sair"
var entrada string = ScanlnStr()
var continuar bool = true
for continuar {
Println(entrada)
entrada = ScanlnStr()
continuar = !(entrada == "sair")
}
Println("Saindo.")
